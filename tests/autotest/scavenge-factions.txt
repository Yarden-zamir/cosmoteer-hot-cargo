# Faction and pirate scavenging (spec 1350 to 1352): an idle pirate collects an old wreck and keeps the loot (no station trades with it);
# an idle faction ship collects, then sells only what it collected and keeps its own stock; the switch turns a side off.
newgame run
wait 8
ev_set Enabled false
godmode on
scav_set FactionEnabled false
# Pirates, far from the players so that no enemy is in sight.
scav_npc pirate 2200
auto_ship Test pirate
auto_wreck 0,250
waitfor scav_wreck_seen == 1 20
scav_age 120
waitfor scav_step == 1 60
waitfor scav_cargo > 0 200
expect scav_sold == 0
# A faction ship.
scav_npc faction 2200
auto_ship Test faction
auto_wreck 0,250
waitfor scav_wreck_seen == 1 20
scav_age 120
waitfor scav_step == 1 60
waitfor scav_excess > 0 200
scav_set FullFraction 0.01
waitfor scav_sold > 0 200
waitfor scav_excess == 0 60
# The switch: with FactionsScavenge off it stops.
scav_set FactionsScavenge false
waitfor scav_step == 0 20
quit
