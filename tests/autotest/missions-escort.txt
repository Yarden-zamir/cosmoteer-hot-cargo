# Escort (spec 36): accept, the freighter steers to the jump point when a player is near, and arrival completes the objective.
newgame run
wait 8
hc_generate escort
hc_accept escort
expect hc_active_escort == 1
# Route (spec 36): MissionTunables.EscortRouteShare (0.65) of the nearest gate distance, so it scales with the system.
expect hc_escort_route_share >= 0.6
expect hc_escort_route_share <= 0.7
expect hc_escort_route >= 7000
expect hc_escort_marker == 1
# Convoy cover: the offer carries MissionTunables.EscortHeatChange, and the reward lowers heat. Heat stays below the first stage, so no hunter disturbs the escort.
expect hc_escort_cover == -8
heat 20
hc_cover_give
expect heat <= 13
hc_escort_goto
# The game can pause by itself here (enemy spotted or a codex page, with default settings in test slots).
pause off
wait 5
expect hc_escort_steering == 1
hc_mark
hc_escort_diag before_flight
# The freighter type is random, and a slow one can take a while to turn and speed up. Wait for progress instead of a fixed time.
waitfor hc_escort_progress > 50 45
hc_escort_diag after_flight
expect hc_escort_done == 0
hc_escort_arrive
wait 3
expect hc_escort_done == 1
# Completion removes the destination marker.
expect hc_escort_marker == 0
quit
