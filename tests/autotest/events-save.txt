# Save and load with a pending event: the page, its serial and the flags come back.
newgame run
wait 8
# The Fence's Ledger page gives the forged codes (fake_codes).
event fence_offer
choose 2 1
expectflag fake_codes 1
heat 30
# The repaint costs 1 money unit (5,000 at tier 1). Leave 1,500 credits, so the repaint is greyed out after the load.
contract_money -6000
event smuggler_den
save events
wait 3
load events
wait 3
expectevent smuggler_den
expectflag fake_codes 1
# Too little money for the repaint: it stays greyed out after the load.
expectchoice 1 disabled
expectchoice 2 enabled
ev_mark
choose 2 1
expect ev_heat_gain <= -29
expect ev_tech_gain == -1
expect ev_pending == 0
expect ev_result >= 2
quit
