# Core: a playthrough without cheats (spec 1004 to 1006). No godmode, no teleports, no test damage, no free money or tech.
# Every command stands in for one player click: an order, a comms button, a trade window transfer, a build confirmation.
# The run passes when the players reach sector 2 with tech and no invariant broke (every ERROR line fails a test).
# timeout: 2100
newgame run
wait 8
pt_watch 30
# Logs only: what enemy ships decide, so a failed run shows where a target went.
tacticlog on
pt_status start
# Ship AI tactics: a preset and the Auto switch, like the tactics editor.
tactics Aggressive
tacticsauto on
# Fly to the nearest station and take the fight with the lowest threat: the lowest tier, then the fewest targets that use
# the Disable tactic (it shoots thrusters and control parts, and the starter ship has no steel to rebuild them), then the
# nearest. The log line "fights offered" shows the choice.
pt_goto station 300
pt_status station
pt_accept
wait 3
expect pt_mission == 1
pt_fight 600
pt_status fight
expect pt_mission == 1
# Back to the station: collect the bounty, sell 1 tech, repair, unlock and build a hyperdrive, buy fuel.
# A ship with shot thrusters still flies, slowly, so the way back may take long.
pt_goto station 900
pt_collect
waitfor pt_mission == 2 30
tech sell 1
wait 20
pt_status sold
# Repair before the long flight to the relay: the crew rebuild shot thrusters from bought steel.
pt_repair 300
pt_status repaired
pt_unlock cosmoteer.hyperdrive_small
waitfor pt_unlocked == 1 30
pt_buy 300 steel 40 coil 90
pt_build cosmoteer.hyperdrive_small
waitfor pt_drives == 1 240
pt_buy 300 hyperium 20
pt_status drive
# The relay toward an unvisited system: fly there, right click it, wait for the drives, press ENGAGE.
pt_goto relay 400
pt_relayjump
waitfor pt_ready == 1 180
fa_engage
waitfor sector >= 2 120
# Headless slots draw no frames, so the jump animation never ends and the sim stays paused (journey-jump.txt).
pause off
wait 5
pt_status sector2
expect sector == 2
expect tech > 0
expect invariants_broken == 0
quit
