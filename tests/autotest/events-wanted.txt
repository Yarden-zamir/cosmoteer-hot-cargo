# Wanted and hunted (spec 470-472): the warning, the Wanted trigger and the events that clear vanilla wanted.
newgame run
wait 8
expectevent getaway
choose 3 1
ev_set WantedEventChance 1
# Wanted alone stays vanilla: no warning, no event. ("wanted" is the UX test command: the nearest station's faction wants you.)
wanted
wait 2
expect ev_wanted == 1
expect ev_wanted_warned == 0
expect ev_pending == 0
# Heat reaches the first stage: the warning fires and a Wanted event opens.
stage
wait 2
expect ev_wanted_warned == 1
expect ev_pending == 1
eventview
expectneed clean_record met
expectneed decoy_run met
# A Clean Record, paid with a crate: wanted clears.
event clean_record
ev_mark
choose 2 1
expect ev_wanted == 0
expect ev_tech_gain == -1
wait 1
expect ev_wanted_warned == 0
expectneed decoy_run unmet
# Wanted again while hunted: the warning fires again. Decoy Run, paid in money: wanted clears.
wanted
wait 2
expect ev_wanted_warned == 1
event decoy_run
ev_mark
choose 1 1
expect ev_wanted == 0
expect ev_money_gain < 0
quit
