# Tech count, single player (spec 65, 442, 449): full-health starting vaults, the damage step, and tech in a derelict ship.
newgame run
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 600
expect vaultminhealth == 3000
expect tech == 8
expect vaultunits == 8
# A small hit costs no tech. Damage past half health costs one unit, in the count and in the storage.
vault damage 0.04
wait 1
expect tech == 8
expect vaultunits == 8
vault damage 0.47
wait 1
expect tech == 7
expect vaultunits == 7
expect paneltotal == 1
# The ship becomes a derelict: the tech still counts, the loss timer stays off, the panel and the blip say where it is.
tech derelict
wait 2
expect tech == 7
expect derelicttech == 7
expect derelicttag == 7
expect panelderelict == 1
expect losstimer == 0
quit
