# Vaults resist sun heat (spec 448): heat damage to a vault × VaultSunHeatFactor. Same dive twice, factor 0 then 1.
newgame run
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 600
worlds jumpto hc_solar_zone
pause off
worlds type hc_solar_zone
# Factor 0: the heat hurts the ship, but not the vaults. Fires that the heat starts still do a little damage (fire is not heat).
tech_set VaultSunHeatFactor 0
worlds goto sun 0.2
wait 3
worlds goto outside
wait 4
expect partdamage > 0
expect vaultdamage < 500
expect tech == 8
worlds cool
vault repair all
# Factor 1: the same dive costs tech.
tech_set VaultSunHeatFactor 1
worlds goto sun 0.2
wait 3
worlds goto outside
wait 4
expect tech < 8
quit
