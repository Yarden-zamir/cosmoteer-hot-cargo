# Join request (spec 1361): a ship carries the page in its comms. The clear deal costs money, and the ship becomes the player's.
newgame run
wait 8
ev_set StationEventChance 0
ev_set ArrivalEventChance 0
ev_set CourierChance 0
ev_set CooldownSeconds 0

expect ev_carrier == 1
ev_carrier_goto

ev_comms carrier Take them in

expect ev_money_gain < 0

wait 5
expect fr_joined == 1
