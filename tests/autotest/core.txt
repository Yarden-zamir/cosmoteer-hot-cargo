# Core heat loop: new career, heat, stages, fame heat, time heat, save and load.
newgame run
wait 8
pu_set Enabled false
log start
expect active == 1
expect sector == 1
# The first stage sets heat to its threshold (Warning, 40).
stage
expect stages == 1
# One standard tier-1 fame unit is 5 fame x 1.5 (default difficulty) x 1.0 (RewardScale) = 7.5 fame. 30 fame = 4 units.
# Run.OnFameChanged rounds the unit to 8 fame, so that is 30 / 8 x 1.425 heat (HeatPerStandardMission) x 1.2 (multiplier with 8 tech) = 6.41 heat.
fame 30
expect heat >= 46.4
expect heat <= 47.6
expect stages == 1
wait 20
expect seconds > 5
save core
wait 3
load core
log loaded
expect heat >= 46.4
expect stages == 1
heat 100
expect stages == stagecount
quit
