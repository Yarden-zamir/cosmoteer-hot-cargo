# Repairs hold in combat (spec 1317): a damaged AI ship that returns to the fight with repairs queued cancels them while an
# enemy within the combat block range fires, as the players' block rule (EnemyCombatNoBuildRadius and Duration).
newgame run
wait 8
ev_set Enabled false
godmode on
tacticlog on
aiset FleeOnVanilla false
aiset RegroupOnVanilla false
aiset CallOnVanilla false
aiset CoverEnabled false
aiset BaitEnabled false
aiset TeamEnabled false
aijitter 0
aitrait Pirate Raiders FallBack 1
aitrait Pirate Ambushers FallBack 1
# The ship returns early, with repairs still queued.
aiset FallBackRecoverFraction 0.4
aiset FallBackMaxSeconds 600
aiset FallBackResupplyEnabled false
openspace 6000
aispawn pirates 1 3 trade
wait 3
aistrip 0 resources
aidamage 0 0.3
aigivelacking 0 0
waitfor ai_fb_active >= 1 30
waitfor ai_new_first_jobs >= 1 120
waitfor ai_fb_active == 0 300
expect ai_new_first_jobs >= 1
# Back to the player ship, which fires at it.
ainew_watch 0 300
waitfor ai_new_first_in_combat == 1 30
waitfor tactic_repair_hold >= 1 10
wait 5
expect ai_new_first_in_combat == 1
expect ai_new_first_jobs == 0
quit
