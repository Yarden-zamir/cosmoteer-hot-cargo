# Surrender (spec 567 to 573): demand that a hunter surrenders (payout, fuel, fuel overflow into space), the once-per-ship rule,
# a hunter's own surrender offer, and the players' surrender to a normal enemy.
newgame run
wait 8
pu_set Enabled false
ev_set Enabled false
godmode on
aiset FleeEnabled false
aiset CallEnabled false
aiset RegroupEnabled false
dip_set DemandHailChance 0
dip_set HunterYieldChance 0
dip_set HunterLeaveSeconds 3
dip_set ExtraLootChance 1
# Refused demand: no surrender, and the same ship takes no second demand even at chance 1.
dip_set SurrenderChanceBase -5
spawnstage Warning
wait 2
dip_near hunter
wait 1
dip_expect demandsurrender hunter 0 == 0
dip demandsurrender hunter
wait 1
expect dip_surrendered == 0
dip_set SurrenderChanceBase 5
dip demandsurrender hunter
wait 1
expect dip_surrendered == 0
# Accepted demand with full fuel tanks: the payout comes in, all hunter fuel drops into space, the hunter jumps out.
dip_fillfuel
dip_mark
spawnstage Warning
wait 2
dip_near fresh
wait 1
dip demandsurrender fresh
wait 1
expect dip_surrendered == 1
# A hunter pays no money (HunterSurrenderMoneyShare 0): only fuel and cargo.
expect dip_moneyspent == 0
expect dip_fuelgain == 0
expect fuelnuggets >= 40
wait 8
expect dip_leaving == 0
# Accepted demand with room: the fuel goes into the tanks.
dip_dropfuel 1000
dip_mark
spawnstage Warning
wait 2
dip_near fresh
wait 1
dip demandsurrender fresh
wait 1
expect dip_surrendered == 2
expect dip_fuelgain >= 40
wait 8
expect dip_leaving == 0
# A beaten hunter offers its surrender and hails; accepting it gives the same loot.
dip_set HunterYieldChance 1
spawnstage Warning
wait 2
dip_near fresh
# The player ship shoots the beaten hunter, so the checks run soon after the hail (sim checks every CheckSeconds).
damagehunters 0.3
wait 1.5
expect dip_yieldoffers >= 1
expect dip_hailing >= 1
dip_dropfuel 1000
dip_mark
dip acceptyield fresh
wait 1
expect dip_surrendered == 3
expect dip_fuelgain >= 1
# The players surrender to a normal enemy: they pay and the enemy stops fighting.
dip_spawn enemy 250
wait 2
expect dip_enemyhostile == 1
dip_page enemy
expect dip_pagebuttons >= 2
dip_set OfferChanceBase 5
dip_mark
dip offersurrender enemy
wait 1
expect dip_moneyspent > 0
expect dip_enemyhostile == 0
quit
