# Event triggers: Arrival (without a jump), Signal POI and Station.
newgame run
wait 8
ev_set CooldownSeconds 0
ev_set ArrivalEventChance 1
ev_set ArrivalDelaySeconds 1
ev_set StationEventChance 0
# No unavoidable page: Vault Coolant Leak has only a submit (spec 811), so "choose last" cannot answer it. events-unavoidable covers it.
ev_set MaxUnavoidablePerRun 0
# No page opens at the start of a run.
expect ev_pending == 0
expect ev_seen == 0
# Arrival: the first-arrival hook opens an Arrival event after the delay.
ev_arrive
wait 4
expect ev_pending == 1
eventview
choose last
expect ev_pending == 0
# Signal: reaching the POI removes it and opens a Signal event.
ev_signal
expect ev_signals >= 1
ev_signal_goto
wait 2
expect ev_pending == 1
eventview
choose last
# Station: the first trade station in a visit rolls once. ev_arrive re-arms the roll.
ev_set StationEventChance 1
ev_set ArrivalEventChance 0
ev_arrive
wait 2
ev_tostation
wait 3
# Some Station event opened (The Fence, The Archive, ...), and the station carries it: its comms show the page (spec 768).
expect ev_pending == 1
expect ev_carrier == 1
choose last
expect ev_pending == 0
quit
