# Rescue Beacon (spec 1147, 1148): a station sells it in its trade window, the fire action shows the fee and heat, and a tug tows the ship to
# the nearest friendly station. Without money the tug comes anyway. A lost tug ends the tow with a message.
newgame run
wait 8
ev_set StationEventChance 0
ev_set ArrivalEventChance 0
ev_set CourierChance 0
godmode on
worlds jumpto hc_trade_hub
ev_tostation
wait 2
# The station stocks beacons at 0.3 money units (times its heat markup), and the player buys one the vanilla way (a trade order).
expect ev_station_beacons >= 1
expect ev_rescue_price_ratio >= 1
expect ev_rescue_price_ratio < 2
expectcomms station Buy a rescue beacon 0
pt_goto station 60
ev_mark
pt_buy 120 yarden_zamir.rescue_beacon 1
expect ev_res_yarden_zamir.rescue_beacon_gain == 1
expect ev_money_gain < 0
# The action button shows while the beacon's part is selected; its tooltip shows the fee.
item_select yarden_zamir.rescue_beacon
expect item_button == 1
expect ev_rescue_tip == 1
# Back to the trade station of the jump, then away from it, so the nearest friendly station is a real tow away. The ship drops
# the move order of the shop trip, so it does not fly back by itself.
ev_stop
ev_tostation
wait 2
ev_tostation 3000
wait 2
# Fire: the fee, the heat, a tug. It reaches the ship, takes it on the line and tows it to the station.
ev_mark
item_action yarden_zamir.rescue_beacon fire
expect ev_res_yarden_zamir.rescue_beacon_gain == -1
expect ev_money_gain < 0
expect ev_heat_gain >= 9
expect ev_rescues == 1
pause off
waitfor ev_rescue_pulling == 1 90
wait 2
expect ev_tug_speed > 1
waitfor ev_rescue_arrivals == 1 300
expect ev_rescues == 0
# No money: the tug takes nothing and comes anyway. Then the tug is lost: the tow ends, the mission stays for the players.
ev_tostation 3000
wait 2
item_give yarden_zamir.rescue_beacon
ev_money_zero
ev_mark
item_action yarden_zamir.rescue_beacon fire
expect ev_money_gain == 0
expect ev_rescues == 1
ev_tug_kill
wait 2
expect ev_rescues == 0
quit
