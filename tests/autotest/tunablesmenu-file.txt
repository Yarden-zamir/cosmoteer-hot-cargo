# A change of tunables.rules while the settings window is open (spec 52, 53, 851, 852): the window shows the new values,
# a bad file keeps the values in effect, and a new fuel floor tops up the live stations at once.
newgame run
wait 8
menu open
tunablesfile Heat { HeatPerGameMinute = 7 }; Balance { StationFuelMinimum = 30 }
wait 3
menu value Heat.HeatPerGameMinute == 7
menu shows Heat.HeatPerGameMinute 7
expect tunables_error == 0
expect balance_station_fuel_floor == 30
expect balance_station_fuel_gap >= 0
# A bad value: rejected, the error shows, and the old values stay.
tunablesfile Heat { HeatPerGameMinute = 9; HeatCarryOverFraction = 2 }
wait 3
expect tunables_error == 1
menu value Heat.HeatPerGameMinute == 7
menu shows Heat.HeatPerGameMinute 7
# A window edit after the file change starts from the file values and writes them back.
menu set Ux.HeatPanel off
expect tunables_error == 0
menu file has HeatPerGameMinute = 7
menu file has StationFuelMinimum = 30
menu file lacks HeatCarryOverFraction
menu close
# tune.sh requests (spec 853): list items and nested paths, all values as one change, and a rejection that changes nothing.
tunerequest set Hunters.Stages[2].ShipCount 5 | set Mechanics.Siege.Seconds 77 | set Heat.HeatMultiplierByUnits 1, 2, 3
wait 2
tuneresult has ok
tuneresult has Hunters.Stages[2].ShipCount: applies NextSpawn
menu value Hunters.Stages[2].ShipCount == 5
menu value Mechanics.Siege.Seconds == 77
menu file has HeatMultiplierByUnits = [1, 2, 3]
tunerequest set Heat.HeatPerGameMinute 3 | set Hunters.Stages[2].ShipCount 0
wait 2
tuneresult has error
tuneresult has ShipCount must be 1 or more
menu value Heat.HeatPerGameMinute == 7
tunerequest set Heat.NoSuchField 1
wait 2
tuneresult has has no tunable 'NoSuchField'
tunerequest reset Hunters.Stages[2].ShipCount | reset Heat.HeatPerGameMinute
wait 2
tuneresult has ok
menu value Hunters.Stages[2].ShipCount == 3
menu value Heat.HeatPerGameMinute == 1.2
quit
