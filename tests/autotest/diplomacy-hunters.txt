# Hunter demands (spec 561 to 566): hand over N, the once-per-ship rule, bluff, deal with 0, 2 and 5 other hunters, the demand hail.
newgame run
wait 8

godmode on
aiset FleeEnabled false
aiset CallEnabled false
aiset RegroupEnabled false
dip_set DemandHailChance 0
dip_set HunterYieldChance 0
dip_set HandOverVisiblePenalty 0
dip_set HandOverPerLeverage 0
expect dip_vaulttech == 8
# The dialogue page of a hunter: the slider and the hand-over, bluff, deal, surrender and refuse buttons.
spawnstage Warning
wait 2
expect hunters == 1
dip_near hunter 250
wait 1
dip_page hunter
# Hand over, bluff and deal each have their own slider in their row (spec 591).
expect dip_pageslider == 3
expect dip_pagebuttons >= 5
# More units, higher chance (no penalty for visible tech here).
dip_expect handover hunter 1 == 0.15
dip_expect handover hunter 7 > 0.8
# A refused hand-over keeps the tech, and the hunter takes no second offer of any kind.
dip_set HandOverChanceMin 0
dip_set HandOverChanceMax 0
dip handover hunter 3
wait 1

expect dip_used >= 4
expect hunters == 1

dip_set HandOverChanceMax 1

wait 1

# An accepted hand-over: the tech goes and the punishment page opens. The wave waits for the answer (spec 594).

wait 2

wait 1
dip handover fresh 3
wait 1
expect dip_vaulttech == 5
expect dip_punishment == 1
expect dip_leaving == 0
dip_page hunter
expect dip_pagetoggles >= 3
# Combos (spec 595): talk alone is low, money and the bribe add, and the sum stops at PunishChanceCap (0.9).
dip_pick talk
wait 1
expect dip_picks == 1
expect dip_punishchance == 0.05
dip_pick money 20
wait 1
expect dip_punishchance == 0.65

dip_unpick money

expect dip_picks == 1
expect dip_punishchance == 0.05

expect heat < 1
wait 8

wait 1

dip bluff hunter 1
wait 1

expect hunters == 0
# Bluff that fails: tech goes, heat rises, the hunter stays.

# A deal that works: pay, keep the tech, the wave leaves.
# A chance above 1 clamps to 1, so the crowd factor of the 5 others does not matter.
