# Hunter demands (spec 561 to 566, 593 to 597): the page with pay off, bluff and fight; pay-off combos, success and failure;
# bluff success and failure; more hunters make a pay-off harder; the demand hail.
newgame run
wait 8
pu_set Enabled false
ev_set Enabled false
godmode on
aiset FleeEnabled false
aiset CallEnabled false
aiset RegroupEnabled false
dip_set DemandHailChance 0
dip_set HunterYieldChance 0
dip_set HunterLeaveSeconds 2
# Fight in open space: a station's faction can take a side (the start station's owner changes with the system layout).
openspace 5000
expect dip_vaulttech == 8
# The page: pay-off rows with their own sliders (tech, money, crew if any can go), then pay off, bluff, demand surrender and fight.
spawnstage Warning
wait 2
expect hunters == 1
dip_near hunter 250
wait 1
dip_page hunter
expect dip_pagetoggles >= 2
expect dip_pageslider >= 2
expect dip_pagebuttons >= 6
# Combos (spec 594): the tech share of PayTechChance (0.9), money adds, the sum stops at PayChanceCap (0.95).
dip_pick hunter tech 4
wait 1
expect dip_picks == 1
expect dip_paychance == 0.45
dip_pick hunter money 20
wait 1
expect dip_paychance == 0.95
dip_pick hunter money 10
wait 1
expect dip_paychance == 0.75
dip_unpick hunter money
wait 1
expect dip_picks == 1
expect dip_paychance == 0.45
# A failed pay-off (spec 596): they keep the offered tech, the money stays, the hunter stays and takes no second offer.
dip_set PayTechChance 0
dip_mark
dip_payoff hunter
wait 1
expect dip_vaulttech == 4
expect dip_moneyspent == 0
expect hunters == 1
expect dip_leaving == 0
expect dip_picks == 0
dip_pick hunter tech 1
wait 1
expect dip_picks == 0
# A pay-off that works (spec 595): the chooser pays, the tech goes, the whole wave leaves, heat drops (spec 593).
dip_set PayTechChance 0.9
dip_set PayMoneyChance 5
dip_set PayChanceCap 1
dip_set HandOverHeatDrop 100
killhunter 0
wait 2
spawnstage Warning
wait 2
dip_near hunter 250
wait 1
heat 20
dip_mark
dip_pick hunter tech 1
dip_pick hunter money 10
dip_pick hunter crew 1
wait 1
expect dip_paychance == 1
dip_log beforepay
dip_payoff hunter
wait 1
expect dip_moneyspent > 0
expect dip_vaulttech == 3
expect dip_leaving >= 1
expect heat < 1
wait 8
expect hunters == 0
# Bluff that works (spec 565): BluffShare of 3 tech is 1; it goes, the wave leaves, heat drops.
dip_set BluffChanceBase 1
dip_set BluffVisiblePenalty 0
dip_set PayMoneyChance 0.6
dip_set PayChanceCap 0.95
spawnstage Warning
wait 2
dip_near hunter 250
wait 1
heat 20
dip_bluff hunter
wait 1
expect dip_vaulttech == 2
expect heat < 1
wait 8
expect hunters == 0
# Bluff that fails: the tech goes, heat rises, the hunter stays.
dip_set BluffChanceBase 0
dip_set BluffPerShare 0
tech add 2
wait 1
spawnstage Warning
wait 2
dip_near hunter 250
wait 1
dip_bluff hunter
wait 1
expect dip_vaulttech == 3
expect hunters == 1
expect heat >= 10
# More hunters near the hailed one make a pay-off harder: × (1 − others ÷ 7) with PayMaxOtherHunters 6, and 0 above the limit.
killhunter 0
wait 2
spawnstage Warning
wait 2
dip_near hunters 300
wait 1
expect hunters == 1
dip_focus hunter
dip_pick focus tech 1
wait 1
expect dip_paychance == 0.3
maxhunters 20
# Two ships of the sector tier: the Warning response of the last sector.
stageset Warning EarlyScale 1
spawnstage Warning
# Back to the default: later Warning spawns send 1 ship in sector 1.
stageset Warning EarlyScale 0.5
wait 2
dip_near hunters 300
wait 1
expect hunters == 3
expect dip_paychance > 0.21
expect dip_paychance < 0.22
spawnstage Hunt
wait 2
dip_near hunters 300
wait 1
expect hunters == 6
dip_set PayMaxOtherHunters 4
expect dip_paychance == 0
# Fight: no cost, the hunter takes no further offer.
dip refuse focus
wait 1
expect dip_vaulttech == 3
expect dip_picks == 0
# The demand hail: a new hunter in comms range hails the players once.
dip_set DemandHailChance 1
dip_set DemandHailGapSeconds 0
spawnstage Warning
wait 2
dip_near hunters 250
wait 3
expect dip_hailing >= 1
quit
