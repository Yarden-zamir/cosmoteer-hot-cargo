# Any "ux: ... failed" line is an ERROR line, so a broken UI patch fails this test.
newgame run
wait 8
# A short loss timer (spec 443), so the loss at the end comes soon after the ship is gone.
tech_set LossGraceSeconds 3
expect active == 1
expect ux_panel == 1
# Two stage starts: the heat panel names each one.
stage
wait 2
stage
wait 2
expect stages == 2
expect ux_stage_alerts == 2
select
damage 0.3
wait 2
expect ux_integrity < 0.9
expect ux_integrity > 0.2
map on
wait 2
map off
ux HeatPanel off
wait 1
expect ux_panel == 0
ux HeatPanel on
closepanel

tostation

comms open
wait 1

comms close
wait 1
expect paused == 0
pause on
wait 1
comms open
wait 1
comms close
wait 1

pause off
