# Telemetry: a short scripted run that makes data for a report (tools/telemetry) and checks that each event kind shows.
# Three systems: the Start event, a contract, an escort, a sale, heat stages with hunters, won fights, a vault hit, an event,
# a save and a load, and a win at the Buyer. It checks telemetry coverage, not balance, so the player's ships cannot be hurt
# (shieldplayers) and damagehostiles stands in for won fights.
newgame run
wait 8
shieldplayers on
# System 1: the Start event, a contract, a sale, an escort, mission fame and heat up to Patrol.
choose 1
hc_generate all
wait 3
contract_money 200000
contract_pick any
contract_accept
tostation
wait 3
tech sell 1
wait 25
hc_accept escort
hc_escort_goto
pause off
wait 5
hc_escort_arrive
fame 15
heat 50
wait 20
damagehostiles 0.05
wait 15
# System 2: a vault hit, an event, a save and a load, Hunt.
jump
waitfor sector >= 2 120
pause off
wait 5
shieldplayers off
vault damage 0.5
shieldplayers on
event distress_beacon
choose 1
heat 75
wait 15
damagehostiles 0.05
save telerun
load telerun
pause off
wait 15
# System 3: the Buyer and the win.
jump
waitfor sector >= 3 120
pause off
wait 5
buyer spawn
wait 3
buyer goto
wait 10
buyer goto
wait 3
buyer sell 2
waitfor won == 1 90
wait 12
teleflush
telecount system.enter >= 2
telecount run.jump >= 2
telecount run.loaded == 1
telecount run.save >= 1
telecount run.won == 1
telecount heat.stage >= 3
telecount heat.accrual >= 3
telecount tech.count >= 3
telecount tech.vault_hit >= 1
telecount tech.value >= 1
telecount econ.money >= 3
telecount econ.fame >= 1
telecount econ.trade >= 2
telecount mission.offered >= 10
telecount mission.accepted >= 2
telecount event.shown >= 2
telecount event.choice >= 2
telecount event.result >= 1
telecount chain.start >= 1
telecount combat.census >= 2
telecount combat.spawn >= 3
telecount combat.damage >= 1
telecount ai.switch >= 1
telecount snap >= 4
telecount perf >= 5
telecount trace >= 5
telecount log >= 20
telestats
expect tele_dropped == 0
expect tele_overhead_pct < 1
quit
