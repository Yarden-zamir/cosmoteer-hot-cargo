# Chains (CHAINS.md section 4): the Salvage Race rivals guard their pod. Before the players accept the stage, they do not attack.
# After the players accept and fly to the pod, the rivals attack.
newgame run
wait 8
expectevent getaway
choose 3 1
ev_set Enabled false
chain_force magpie_rivalry
expectstage race
chain_tostation
expect partdamage == 0
expect vaultdamage == 0
# 30 s at the station with the offer open: the rivals stay at the pod.
wait 30
expect chain_accepted == 0
expect chain_rival_dist > 2000
expect partdamage == 0
expect vaultdamage == 0
# Accept, then fly to the pod: the rivals attack.
chain_accept
expect chain_accepted == 1
chain_goto hc_ch_pod
waitfor partdamage > 0 45
quit
