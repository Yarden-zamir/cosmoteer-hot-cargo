# Run modes, fame title prototype (spec 686). Off by default: fame changes no hunter tier. On: a fame title above the system tier adds
# tiers to a real stage spawn, up to FameTierCap.
newgame run
wait 8
ev_set Enabled false
godmode on
maxhunters 0
rm_missions 6
expect rm_fametier >= 2
expect rm_famebonus >= 1
expect rm_tiers == 0
rm_set FameTitleTiers on
expect rm_tiers >= 1
rm_set FameTierCap 0
expect rm_tiers == 0
rm_set FameTierCap 2
maxhunters 12
spawnstage Patrol
wait 2
rm_expect_stage Patrol
quit
