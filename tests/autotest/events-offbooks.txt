# Off the Books through the real comms window (spec 767-771): the line opens the page in the same tab, and its choices work.
newgame run
wait 8
ev_set StationEventChance 0
ev_set CourierChance 0
ev_set CooldownSeconds 0
save offbooks
# 1. Sell one crate (an option row, spec 810): tech drops by 1, money rises, the page closes.
ev_open station
ev_click station Sell a crate off the books
expectevent off_books
# Spec 816: the window is on the screen and shows the whole page or its scroll bar.
expect comms_fit == 1
ev_mark
pick 1
expectoption 1 selected
ev_click station Sell the crates
expect ev_tech_gain == -1
expect ev_money_gain > 0
expect ev_pending == 0
# 2. Changed my mind: the page closes with no change.
load offbooks
wait 3
ev_open station
ev_click station Sell a crate off the books
expectevent off_books
ev_mark
ev_click station Changed my mind
expect ev_pending == 0
expect ev_tech_gain == 0
expect ev_money_gain == 0
quit
