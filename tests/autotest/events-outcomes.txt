# Events (EVENTS.md): forced events, choices and their effects on money, heat, tech, flags and spawns.
newgame run
wait 8
# No page opens at the start of a run.
expect ev_pending == 0
# Every event loads, validates and renders with the starting fleet.
eventviewall
# Fence: tech for money and less heat. The sale is a multiselect page (spec 810): pick the crates, then submit.
heat 40
event the_fence
eventview
expect ev_stdmoney > 0
expectchoice 1 disabled
expectchoice 2 enabled
pick 1
expectoption 1 selected
expectchoice 1 enabled
ev_mark
choose 1 1
expect ev_tech_gain == -1
expect ev_heat_gain <= -19
expect ev_money_gain > 0
# Two crates: heat -20 for the first and -15 for the second, as the old two-crate choice.
heat 60
event the_fence
pick 1 2
ev_mark
choose 1 1
expect ev_tech_gain == -2
expect ev_heat_gain <= -34
expect ev_heat_gain >= -36
# Checkpoint: the forged-codes option stays hidden without the flag; the submit waits for an option.
event corporate_checkpoint
expectoption 3 hidden
expectoption 1 enabled
expectchoice 1 disabled
expectchoice 2 enabled
# A caught scan leads to a follow-up page (Next).
choose 2 2
expectevent checkpoint_caught
eventview
# Breaking away spawns a corporate patrol with the hunter AI.
ev_mark
choose 2 1
expecttagged hc_ev_corp >= 2
expect ev_heat_gain >= 19
expect ev_pending == 0
# Forged codes from the Fence's Ledger page: the flag shows the hidden choice, and using it clears the flag.
event fence_offer
choose 2 1
expectflag fake_codes 1
event corporate_checkpoint
expectoption 3 enabled
pick 3
ev_mark
choose 1
expectflag fake_codes 0
expect ev_heat_gain <= -19
# Resources: a derelict salvage adds steel; hull damage hits parts.
event derelict_survey
ev_mark
choose 1 1
expect ev_res_steel_gain >= 40
event derelict_survey
choose 2 2
# Magpies: the fight starts a mission with a bounty.
event rival_crew
choose 3 1
expecttagged hc_ev_magpie >= 1
# Crew: the beacon engineer joins when a bunk is free.
event distress_beacon
ev_mark
choose 1 2
expect ev_crew_gain <= 1
# The MP input path with a real roll.
event odd_merchant
choose last
expect ev_pending == 0
quit
