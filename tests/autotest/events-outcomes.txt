# Events (EVENTS.md): forced events, choices and their effects on money, heat, tech, flags and spawns.
newgame run
wait 8
# The Start event opens StartDelaySeconds (3 game s) after a new run.
expectevent getaway
# Every event loads, validates and renders with the starting fleet.
eventviewall
eventview
expect ev_stdmoney > 0
expectchoice 1 enabled
expectchoice 2 enabled
# Getaway, choice 2: sell one crate for 1.2 x the tech value.
ev_mark
choose 2 1
expect ev_tech_gain == -1
expect ev_money_gain >= 11000
expect ev_pending == 0
expect ev_result >= 1
eventview
# Fence: tech for money and less heat.
heat 40
event the_fence
expectchoice 1 enabled
expectchoice 2 enabled
ev_mark
choose 1 1
expect ev_tech_gain == -1
expect ev_heat_gain <= -19
expect ev_money_gain > 0
# Checkpoint: the forged-codes choice stays hidden without the flag.
event corporate_checkpoint
expectchoice 1 hidden
expectchoice 2 enabled
# A caught scan leads to a follow-up page (Next).
choose 3 2
expectevent checkpoint_caught
eventview
# Breaking away spawns a corporate patrol with the hunter AI.
ev_mark
choose 2 1
expecttagged hc_ev_corp >= 2
expect ev_heat_gain >= 19
expect ev_pending == 0
# Forged codes: the flag shows the hidden choice, and using it clears the flag.
event getaway
choose 1 1
expectflag fake_codes 1
event corporate_checkpoint
expectchoice 1 enabled
ev_mark
choose 1 1
expectflag fake_codes 0
expect ev_heat_gain <= -19
# Resources: a derelict salvage adds steel; hull damage hits parts.
event derelict_survey
ev_mark
choose 1 1
expect ev_res_steel_gain >= 40
event derelict_survey
choose 2 2
# Magpies: the fight starts a mission with a bounty.
event rival_crew
choose 4 1
expecttagged hc_ev_magpie >= 1
# Crew: the beacon engineer joins when a bunk is free.
event distress_beacon
ev_mark
choose 1 2
expect ev_crew_gain <= 1
# The MP input path with a real roll.
event odd_merchant
choose last
expect ev_pending == 0
quit
