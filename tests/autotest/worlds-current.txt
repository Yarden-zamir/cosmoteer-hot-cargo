# Current (spec 221): a ship at rest in the current drifts along the flow; beside the current it stays at rest.
# timeout: 900
# Ships and loose items get the same push: along the flow and toward the nearer side, so drifting things leave the band.
newgame run
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 900
worlds jumpto hc_trade_hub
pause off
expect worlds_current_tris >= 6
expect worlds_current_tris <= 20
worlds goto nocurrent
expect worlds_in_current == 0
wait 4
expect worlds_flow_speed < 1
expect worlds_flow_speed > -1
worlds goto current
expect worlds_in_current > 0.5
expect worlds_ship_in_current > 0.5
# A tech crate dropped near the band center floats with the current.
tech drop 1
wait 1
expect worlds_items_in_current >= 1
expect worlds_crate_along < 1500
expect worlds_crate_across < 600
expect worlds_sim_seconds > 0
wait 4
expect worlds_item_flow_speed > 5
# An idle ship brakes against the drift, but it keeps drifting downstream.
wait 5
expect worlds_flow_speed > 2
# The crate leaves the band. With the side share 0.25, drift from near the axis leaves near the far end (half length 3000),
# and it moves away from the axis on the way out.
waitfor worlds_items_in_current == 0 300
expect worlds_sim_seconds > 0
expect worlds_crate_along > 2500
expect worlds_crate_across > 500
waitfor worlds_ship_in_current < 0.01 800
expect worlds_sim_seconds > 0
quit
