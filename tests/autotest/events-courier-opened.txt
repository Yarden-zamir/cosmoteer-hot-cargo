# Spec 1117, 1118: the courier contact takes an opened case and reacts (forced rolls). It pays 20 % less, turns hostile, or jumps
# away. Hostile and gone fail the job; the case is gone either way.
newgame run
wait 8
godmode on
ev_set CourierChance 0
ev_set StationEventChance 0
ev_set ArrivalEventChance 0
worlds jumpto hc_trade_hub
# Job 1: the contact notices the seal and pays 20 % less.
ev_tostation
wait 2
ev_courier_offer
ev_courier_accept
expect ev_courier_job == 0
ev_mark
item_force 1
item_action yarden_zamir.sealed_case open
expect ev_res_yarden_zamir.opened_case_gain == 1
expect ev_courier_job == 0
# The first teleport loads the contact out of stasis, the second one puts the ships next to its live position.
ev_courier_goto
pause off
ev_courier_goto
pause off
waitfor ev_courier_seen == 1 20
ev_contact_mark
ev_courier_pay_mark
ev_comms contact Hand over the case.
expectevent courier_handover
ev_force_outcome 2
ev_comms carrier Hand over the opened case.
expect ev_courier_job == 1
expect ev_res_yarden_zamir.opened_case_gain == 0
expect ev_courier_pay_ratio > 0.75
expect ev_courier_pay_ratio < 0.85
# Job 2: the contact turns hostile. The job fails.
ev_tostation
wait 2
ev_courier_offer
ev_courier_accept
expect ev_courier_job == 0
ev_mark
item_force 1
item_action yarden_zamir.sealed_case open
expect ev_res_yarden_zamir.opened_case_gain == 1
expect ev_courier_job == 0
# The first teleport loads the contact out of stasis, the second one puts the ships next to its live position.
ev_courier_goto
pause off
ev_courier_goto
pause off
waitfor ev_courier_seen == 1 20
ev_contact_mark
ev_courier_pay_mark
ev_comms contact Hand over the case.
expectevent courier_handover
ev_force_outcome 3
ev_comms carrier Hand over the opened case.
expect ev_courier_job == -1
expect ev_contact_hostile == 1
expect ev_res_yarden_zamir.opened_case_gain == 0
# Job 3: the contact jumps away. The job fails.
ev_tostation
wait 2
ev_courier_offer
ev_courier_accept
expect ev_courier_job == 0
ev_mark
item_force 1
item_action yarden_zamir.sealed_case open
expect ev_res_yarden_zamir.opened_case_gain == 1
expect ev_courier_job == 0
# The first teleport loads the contact out of stasis, the second one puts the ships next to its live position.
ev_courier_goto
pause off
ev_courier_goto
pause off
waitfor ev_courier_seen == 1 20
ev_contact_mark
ev_courier_pay_mark
ev_comms contact Hand over the case.
expectevent courier_handover
ev_force_outcome 4
ev_comms carrier Hand over the opened case.
expect ev_courier_job == -1
waitfor ev_contact_gone == 1 20
expect ev_res_yarden_zamir.opened_case_gain == 0
quit
