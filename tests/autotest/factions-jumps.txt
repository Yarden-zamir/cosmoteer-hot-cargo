# Factions: relay and blind jumps (spec 650, 651). Every jump cuts heat: a relay with hunters near it by RelayWatchedHeatCut (0.75),
# a relay with no hunter near it by RelayClearHeatCut (0.85), a blind jump by BlindHeatCut (0.9). A blind jump costs BlindFuelFactor
# more fuel and can damage parts on arrival. The blind switch holds for one jump.
# Headless slots draw no frames, so the sim stays paused after a jump (see journey-jump.txt): one jump per game.
# Time heat after the arrival adds up to about 1 in the waits.

# Clear relay, and the blind fuel. The hunters spawn near the fleet at the start, far from the relays.
newgame
wait 8
ev_set Enabled false
godmode on
fa_drive
wait 1
expect fa_fuel > 0
fa_base
expect fa_blind == 0
fa_blind on
wait 1
expect fa_blind == 1
expect fa_fuelratio > 1
expect fa_blindfuelgap == 0
fa_tooltip
fa_blind off
wait 1
expect fa_fuelratio == 1
heat 80
wait 2
expect fa_huntersnear == 0
fa_jump relay
waitfor sector == 2 60
expect heat >= 11.5
expect heat <= 13.5

# Watched relay: the fleet waits at the relay, so the hunters spawn near it. 80 x 0.25 = 20.
newgame
wait 8
ev_set Enabled false
godmode on
fa_torelay
wait 2
heat 80
wait 2
expect fa_huntersnear >= 1
fa_tooltip
fa_jump relay
waitfor sector == 2 60
expect heat >= 19.5
expect heat <= 21.5

# Blind: 80 x 0.1 = 8. With chance 1 the ship arrives with damaged parts. The switch turns off after the jump.
# No godmode here: it heals the parts.
newgame
wait 8
ev_set Enabled false
fa_set BlindDamageChance 1
fa_blind on
wait 1
heat 80
expect fa_hurtparts == 0
fa_jump blind
waitfor sector == 2 60
expect heat >= 7.5
expect heat <= 9.5
expect fa_blind == 0
expect fa_hurtparts >= 1
quit
