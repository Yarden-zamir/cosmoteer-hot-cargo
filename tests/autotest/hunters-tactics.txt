# Hunter AI tactics: modules load, act, and log. Strike stage turns on every tactic (Disable at Hunt, Bait at Strike).
newgame run
wait 8
# No random events during the hunter checks.
ev_set Enabled false
# Nine strike-stage hunters destroy a sector-1 starter ship before the tactics can be observed, even with godmode
# (one big hit still destroys a part). Five hunters are enough for groups (pincer, bait) and every stage (disable, bait).
godmode on
maxhunters 5
tacticlog on
# The enemy AI behaviors (flee, regroup) have their own tests; here they would change the tactic counts.
aiset FleeEnabled false
aiset RegroupEnabled false
# Every hunter uses every tactic, as before personalities (spec 360).
aiset PersonalitiesEnabled false
heat 100
wait 1
hunters start
expect hunters == 5
expect stages == 4
wait 45
hunters engaged
expect tactic_hunt > 0
expect tactic_pincer > 0
expect tactic_bait > 0
expect tactic_disable > 0
log engaged
damagehunters 0.2
wait 6
hunters damaged
expect tactic_fallback > 0
quit
