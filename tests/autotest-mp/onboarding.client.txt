# Co-op onboarding, client side. "mp stage" starts the next heat stage in the sim on both machines; each machine logs the stage itself.
codex_reset
tutorials on
mpjoin 127.0.0.1:{PORT}
mpready
wait 8
expect hint_hc_coop == 1
waitfor mpmark >= 1 60
wait 3
expect runlog_stage == 1
expect runlog_count >= 2
expect desyncs == 0
codex_restore
quit
