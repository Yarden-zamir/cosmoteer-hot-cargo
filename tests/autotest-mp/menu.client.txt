# Co-op settings menu, client side: the host's values arrive, and the client's window is read-only.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor heatperminute == 45 90
menu open
expect menu_readonly == 1
menu reject Heat.HeatPerGameMinute 10
menu close
menu value Hunters.Stages[0].ShipCount == 2
menu value Heat.HeatCarryOverFraction == 0.5
mp mpmark 1
waitfor hunters >= 2 60
wait 20
expect desyncs == 0
log client-end
quit
