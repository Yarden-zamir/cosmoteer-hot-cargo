# Co-op factions prices, client side. "tech sell" and "fa_buy" send the game's own trade inputs from the client.
mpjoin 127.0.0.1:{PORT}
mpready
mp tostation
wait 3
mp mpmark 1
waitfor mpmark >= 2 120
expect fa_rivalsell == 1.5
# One unit to the rival's station pays the fence premium (tech value 15,000 at the start, x1.5).
tech sell 1
waitfor moneygain >= 22000 120
mp mpmark 3
waitfor mpmark >= 4 120
expect fa_markup == 1.2
expect fa_buyratio == 1.2
fa_buy 5
waitfor fa_spent > 0 60
expect fa_buygap == 0
mp mpmark 5
waitfor mpmark >= 5 30
wait 3
expect desyncs == 0
quit
