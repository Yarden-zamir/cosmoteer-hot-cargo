# Co-op ship tactics, client side.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor active == 1 60
tactics Sniper
tacticsauto on
waitfor tactics_any_auto >= 2 90
expecttacticsany Defensive
# Editor edits: apply a preset, change the action of row 1, move row 3 up. Each edit sends the rows as one input.
tactics_editor open
tactics_editor apply Escort
tactics_editor act 1 RetreatRepair
tactics_editor up 3
wait 3
expecttactics HC1:HealthBelow=25>RetreatRepair;Always>Guard=0;AllyUnderAttack>TargetAllyAttacker
expect tactics_auto == 1
tactics_editor close
mp mpmark 1
waitfor mpgames >= 2 180
waitfor active == 1 60
expecttacticsany Defensive
expecttactics HC1:HealthBelow=25>RetreatRepair;Always>Guard=0;AllyUnderAttack>TargetAllyAttacker
expect tactics_auto == 1
waitfor mpmark >= 2 120
expect tactics_evals > 20
expect desyncs == 0
quit
