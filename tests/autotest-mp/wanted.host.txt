# Co-op wanted and hunted (spec 470-472), host side: both machines become wanted and hunted, the Wanted event opens on both,
# the client pays a crate for a clean record, and wanted clears on both machines with the same state.
mphost {PORT}
mpstart 2
mp pu_set Enabled false
waitfor ev_pending == 1 30
expectevent getaway
choose 3
waitfor ev_pending == 0 30
mp ev_set WantedEventChance 1
mp wanted
wait 2
expect ev_wanted == 1
mp heat 30
waitfor ev_pending == 1 20
expect ev_wanted_warned == 1
mp event clean_record
waitfor ev_wanted == 0 90
expect ev_pending == 0
expect desyncs == 0
waitfor mpconnected == 0 240
quit
