# Co-op Pursuit, client side: the stage and the hacker come through mp commands; the payment is the client's own shared choice input.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor mpmark >= 1 60
mp heat 25
waitfor pu_scouts == 1 30
mp pu_scoutnear 300
waitfor pu_scouts == 0 60
expect heat >= 39.5
mp pu_hacker
waitfor pu_hacker == 1 30
mp pu_addmoney 50000
mp pu_hacker_near 400
wait 3
pu_money
pu_hacker_pick pay
waitfor pu_outage >= 200 30
expect pu_moneyspent == 30000
expect pu_hacker_done == 1
wait 10
expect desyncs == 0
log client-end
quit
