# Co-op shared choices, client side.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor ev_pending == 1 30
hover 3
# The host saw the hover.
waitfor mpmark >= 1 60
# A steady hover sends a heartbeat every PresenceExpirySeconds / 3 (1 s), never every frame.
sc_presence_mark
wait 10
expect sc_presence_sent <= 15
expect sc_presence_sent >= 5
hover off
waitfor sc_votemode == 1 60
vote 3
waitfor sc_decided_reason == 1 60
expect sc_decided_choice == 3
# Timeout round.
waitfor ev_pending == 1 60
ev_mark
vote 2
@2p waitfor sc_decided_reason == 3 60
@3p waitfor sc_decided_reason == 2 60
expect sc_decided_choice == 2
expect ev_tech_gain == -1
# Force round. A pick in vote mode does nothing; the vote of the first client counts.
waitfor ev_pending == 1 60
choose 1
wait 3
expect ev_pending == 1
@c1 vote 3
waitfor sc_decided_reason == 4 60
expect sc_decided_choice == 3
expect ev_pending == 0
expect desyncs == 0
quit
