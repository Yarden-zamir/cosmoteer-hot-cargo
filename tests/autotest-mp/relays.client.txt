# Co-op relay lockdown, client side: the client moves the guards next to its ship and selects credits on the bribe page.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor mpmark >= 1 120
mp rl_guardsnear
waitfor rl_live >= 1 15
wait 2
rl_pick money price
waitfor rl_picks >= 1 30
waitfor rl_blocks == 0 90
expect rl_guards == 0
wait 10
expect desyncs == 0
log client-end
quit
