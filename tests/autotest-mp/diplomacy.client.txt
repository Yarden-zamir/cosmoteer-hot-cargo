# Co-op diplomacy, client side: the client's own choices, sent as real shared choice inputs.
# With 3 players, the first client acts and the second client checks the same results.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor mpmark >= 1 60
@c1 mp spawnstage Warning
waitfor hunters >= 1 30
@c1 mp dip_near hunters
wait 2
@c1 dip handover hunter 2
waitfor dip_vaulttech == 6 30
waitfor dip_punishment == 1 30
@c1 dip_pick talk
waitfor dip_picks >= 2 60
@c1 dip_punish submit
waitfor dip_punishment == 0 30
waitfor hunters == 0 60
@c1 mp dip_spawn friend 250
wait 3
@c1 dip hire friend 0
waitfor dip_friendhired == 1 30
wait 10
expect desyncs == 0
log client-end
quit
