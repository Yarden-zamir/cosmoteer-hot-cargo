# Co-op autonomous ships, host side (spec 1075, 1082): a looter and a miner of the host run on both machines. Funnel pickups,
# Computer manning, mining, the instant sale and the money must match: any desync fails the test.
mphost {PORT}
mpstart 2
mp ev_set Enabled false
mp godmode on
mp auto_build Miner player yarden_zamir.computer@2,0,1 cosmoteer.mining_laser_small@3,0,1 yarden_zamir.targeted_reactor@2,2 cosmoteer.storage_2x2@0,0 yarden_zamir.funnel@0,2 cosmoteer.thruster_small@0,4 cosmoteer.thruster_small@3,4 cosmoteer.thruster_small@0,-2,2 cosmoteer.thruster_small@3,-2,2 cosmoteer.thruster_small@-2,0,1 cosmoteer.thruster_small@-2,2,1 cosmoteer.thruster_small@6,0,3 cosmoteer.thruster_small@4,2,3
wait 1
tactics_ship Miner
tactics HC1:CargoAbove=25>GoToStation=0;StationInRange&CargoToSell>Sell;LootWithin=600>CollectLoot;AsteroidWithin=20000>Mine;Always>ReturnToFleet
tacticsauto on
mp auto_build Looter away yarden_zamir.computer@2,0 yarden_zamir.targeted_reactor@2,1 cosmoteer.storage_2x2@0,0 yarden_zamir.funnel@0,2 cosmoteer.thruster_small@0,4 cosmoteer.thruster_small@3,3 cosmoteer.thruster_small@0,-2,2 cosmoteer.thruster_small@3,-2,2 cosmoteer.thruster_small@-2,0,1 cosmoteer.thruster_small@-2,2,1 cosmoteer.thruster_small@4,0,3 cosmoteer.thruster_small@4,1,3
wait 1
tactics_ship Looter
tactics Looter
tacticsauto on
mp auto_mark
mp auto_loot gold 1 24 120
auto_ship Looter
waitfor auto_money_gain > 0 400
auto_ship Miner
waitfor auto_salvage_jobs >= 1 300
waitfor auto_cargo >= 1 300
wait 20
expect desyncs == 0
mp mpmark 1
waitfor mpconnected == 0 240
quit
