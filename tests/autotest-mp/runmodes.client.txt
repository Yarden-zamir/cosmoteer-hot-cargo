# Co-op run modes, client side. The client sees the host's levels and the same effective values.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor mpmark >= 1 90
expect rm_heatrate > 1
expect rm_ships > 1
expect rm_tech < 1
expect rm_techgap < 0.01
expect pricegap == 0
rm_log client
waitfor mpmark >= 2 90
expect rm_tiers >= 1
waitfor hunters >= 3 30
rm_log client-after
hunters client
wait 30
expect desyncs == 0
log client-end
quit
