# Co-op research trade (spec 760-763), host side: the offer is built the same on both machines, the client trades, and the
# techs unlock on both with no desync.
mphost {PORT}
mpstart {PLAYERS}
waitfor ev_pending == 1 30
expectevent getaway
choose 3
waitfor ev_pending == 0 30
ev_researched_mark
mp event research_trade
waitfor ev_pending == 1 20
expect ev_offer_rate >= 1.2
waitfor ev_pending == 0 90
expect ev_researched_gain >= 1
expect desyncs == 0
waitfor mpconnected == 0 240
quit
