# Co-op contract, client side. The "mp" lines run in the sim on both machines; contract_accept sends the game's accept input from the client.
mpjoin 127.0.0.1:{PORT}
mpready
wait 5
mp contract_money 500000
wait 2
mp contract_pick paid
wait 5
mp contract_terms informant 0
wait 2
contract_accept
waitfor contract_accepted == 1 30
expect contract_fee_diff == 0
expect contract_bonus_payer_local == 1
mp contract_give_bonus
wait 3
expect contract_bonus_given == 1
expect contract_money_change >= 1000
expect desyncs == 0
quit
