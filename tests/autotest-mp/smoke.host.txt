# Co-op smoke test, host side: host a Hot Cargo career, start it with one client, check that both play the same run.
mphost {PORT}
mpstart 2
waitfor active == 1 60
log host-start
expect ishost == 1
expect mpplayers == 2
wait 30
expect desyncs == 0
# The client quits first. Wait until it is gone, so the client never sees the host leave.
waitfor mpconnected == 0 120
quit
