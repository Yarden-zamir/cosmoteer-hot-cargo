# Co-op save, resync and load, client side.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor stages >= 1 60
waitfor ev_pending == 1 30
waitfor mpgames >= 2 180
waitfor active == 1 60
expectevent smuggler_den
expect hc_trap_state == 2
expect stages >= 1
mp mpmark 1
waitfor mpsetup == 1 120
mpready
expectevent smuggler_den
expect hc_trap_state == 2
expect stages >= 1
expect heat >= 30
waitfor mpmark >= 2 60
expect desyncs == 0
quit
