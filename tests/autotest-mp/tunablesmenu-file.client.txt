# Co-op change of the host's tunables.rules, client side: the host's values arrive, and the client's own file has no effect.
mpjoin 127.0.0.1:{PORT}
mpready
waitfor heatperminute == 45 90
menu open
expect menu_readonly == 1
menu shows Heat.HeatPerGameMinute 45
menu close
menu value Hunters.MaxLiveHunters == 7
tunablesfile Heat { HeatPerGameMinute = 99 }
wait 3
expect heatperminute == 45
mp mpmark 1
waitfor hunters >= 1 60
wait 20
expect desyncs == 0
log client-end
quit
