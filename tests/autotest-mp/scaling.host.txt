# Co-op enemy scaling, host side (spec 330 to 336). The host's slider value (2.0) reaches the client, whose own value is 1.0.
# With 2 players the factor is 2: both machines generate the first system with it (MPSTATE compares the ship counts),
# and a hunter wave and an event group get twice the ships on both machines, with no desync.
enemysetting 2.0
mphost {PORT}
mpstart 2
mp pu_set Enabled false
expect enemysetting == 2
expect enemyfactor == 2
mp godmode on
mp ev_set Enabled false
# Two ships of the sector tier: the Warning response of the last sector.
mp stageset Warning EarlyScale 1
mp spawnstage Warning
waitfor hunters >= 4 30
expect hunters == 4
mp ev_spawn TollGuards
mp counttag hc_ev_toll
waitfor tagcount >= 6 30
expect tagcount == 6
# The first system was generated with factor 2: pirate wanderers are [9, 11] (mod/contracts/actions.rules), so 18 to 22.
counttag pirate
expect tagcount >= 18
wait 30
expect desyncs == 0
log host-end
waitfor mpconnected == 0 240
quit
