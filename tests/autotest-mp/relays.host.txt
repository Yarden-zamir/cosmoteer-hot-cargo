# Co-op relay lockdown, host side (spec 958, 960): pirates hold the relays. The client builds the bribe draft, the host submits it
# through the shared choice input. Both machines see the relays open and the guards stand down, with no desync.
mphost {PORT}
mpstart 2
mp godmode on
mp ev_set Enabled false
mp contract_money 100000
mp rl_here pirates
waitfor rl_guards >= 2 60
mp mpmark 1
waitfor rl_picks >= 1 120
wait 2
rl_bribe
waitfor rl_blocks == 0 60
expect rl_guards == 0
wait 10
expect desyncs == 0
log host-end
waitfor mpconnected == 0 240
quit
