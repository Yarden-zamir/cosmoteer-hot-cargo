# Co-op loss timer, host side (spec 443): tech at 0 starts the timer on both machines, tech back stops it, and the timer ends the run.
tunablesfile Tech { LossGraceSeconds = 30 }  Events { Enabled = false }
mphost {PORT}
mpstart 2
waitfor mpmark >= 1 120
# All tech goes out as crates: the timer starts on both machines, and both see the crate markers.
mp tech drop 8
waitfor losstimer == 1 30
expect techcrates == 8
expect losstimerleft <= 30
mp mpmark 2
# The client gets tech back.
waitfor losstimer == 0 60
expect lost == 0
mp tech remove 2
waitfor losstimer == 1 30
waitfor lost == 1 60
expect desyncs == 0
waitfor mpconnected == 0 240
quit
