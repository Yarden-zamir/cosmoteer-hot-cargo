# Co-op run modes, host side (spec 680 to 685). The host changes the difficulty through the vanilla settings inputs. Both machines
# then use the same effective heat rate, hunter count and tier, tech value and fees, with no desync.
mphost {PORT}
mpstart 2
mp godmode on
mp ev_set Enabled false
rm_difficulty very_hard slow scarce
rm_log host
mp mpmark 1
# Heat from time at the effective rate on both machines (MPSTATE compares the heat).
wait 40
# Two ships of the sector tier: the Warning response of the last sector.
mp stageset Warning EarlyScale 1
mp spawnstage Warning
waitfor hunters >= 3 30
mp mpmark 2
rm_log host-after
hunters host
wait 30
expect desyncs == 0
log host-end
waitfor mpconnected == 0 240
quit
