# Co-op trade and loss, host side. Each player's starter ship has 4 vaults (spec 1011) and half of the tech (4 + 4, spec 17).
# Each player sells from their own ship to a station and gets the money. The last sale loses the run for both.
mphost {PORT}
mpstart 2
mp ev_set Enabled false
# A short loss timer (spec 443), so the last sale loses the run within the test.
mp tech_set LossGraceSeconds 3
expect vaulttech == 8
expect vaults == 8
expect vaultpaths == 8
expect noaccess == 0
# The host money before the client sells: its sale must not change it.
hc_mark
mp tostation
# The client sells its 4 units first.
waitfor tech == 4 120
expect hc_money_gain == 0
tech sell 4
waitfor lost == 1 120
expect tech == 0
expect moneygain >= 60000
expect desyncs == 0
waitfor mpconnected == 0 240
quit
