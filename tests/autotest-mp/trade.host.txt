# Co-op trade and loss, host side. Each player's starter ship has 4 vaults (spec 1011) and a share of the tech (spec 17):
# 4 + 4 with 2 players, 3 + 3 + 2 with 3 players.
# Each player sells from their own ship to a station and gets the money. The last sale loses the run for both.
mphost {PORT}
mpstart {PLAYERS}
mp ev_set Enabled false
# A short loss timer (spec 443), so the last sale loses the run within the test.
mp tech_set LossGraceSeconds 3
expect vaulttech == 8
@2p expect vaults == 8
@2p expect vaultpaths == 8
@3p expect vaults == 12
@3p expect vaultpaths == 12
expect noaccess == 0
# The host money before the client sells: its sale must not change it.
hc_mark
mp tostation
# The clients sell their units first.
@2p waitfor tech == 4 120
@3p waitfor tech == 3 120
expect hc_money_gain == 0
@2p tech sell 4
@3p tech sell 3
waitfor lost == 1 120
expect tech == 0
@2p expect moneygain >= 60000
@3p expect moneygain >= 45000
expect desyncs == 0
waitfor mpconnected == 0 240
quit
