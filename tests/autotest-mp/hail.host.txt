# Co-op ship comms (spec 771), host side: a freighter carries the page; the client chooses in its comms; the host's late click on
# the same page (same serial) does nothing, so the choice applies once on both machines.
mphost {PORT}
mpstart {PLAYERS}
waitfor ev_pending == 1 30
expectevent getaway
choose 3
waitfor ev_pending == 0 30
mp ev_set StationEventChance 0
mp event broken_down
waitfor ev_carrier == 1 20
ev_mark
waitfor ev_pending == 0 90
# The client stripped the cargo. A late click on the same page (serial 2) arrives now.
mpchoose 2 3
wait 3
expect ev_res_steel_gain == 30
expect desyncs == 0
waitfor mpconnected == 0 240
quit
