# Co-op session: friendly encounters (spec 1360 and later). See coop-core.txt for "client:", "as 1", "on 1" and steps.
# Each encounter is a ship-carried event page; the choices go through the shared choice input, and the effects run in the sim.
mphost {PORT}
mpstart {PLAYERS}
mp godmode on
mp ev_set Enabled false
closepanel
on 1 closepanel

step friendly.join
# Spec 1361: the client takes the join request from its machine. The ship becomes the client's on both machines; the client pays.
mp event join_request
waitfor ev_pending == 1 20
mp ev_carrier_goto
wait 2
ev_mark
on 1 ev_mark
on 1 ev_comms carrier Take them in
waitfor ev_pending == 0 30
on 1 waitfor ev_pending == 0 30
on 1 expect fr_player_ships == 2
on 1 expect fr_joined == 1
on 1 expect ev_money_gain < 0
expect fr_player_ships == 1
expect ev_money_gain == 0

step friendly.apprentice
# Spec 1362, 1363: the client lets an apprentice follow. Its hire and count run in the sim on both machines; at its time,
# with no fights seen, it leaves on both.
mp event apprentice
waitfor ev_pending == 1 20
mp ev_carrier_goto
wait 2
on 1 ev_comms carrier Let them follow
waitfor fr_apprentices == 1 30
on 1 waitfor fr_apprentices == 1 30
expect fr_apprentice_hired == 1
on 1 expect fr_apprentice_hired == 1
mp fr_apprentice_end
waitfor fr_apprentices == 0 20
on 1 waitfor fr_apprentices == 0 30

step friendly.trader
# Spec 1364: the client selects a trade in the shared draft and submits it. The client pays its steel; the processor arrives on
# both machines' view of the client's ship; the trader leaves on both.
mp event wandering_trader
waitfor ev_pending == 1 20
mp ev_carrier_goto
as 1 ev_storage_fill steel
wait 2
ev_mark
on 1 ev_mark
on 1 pick 1 1
on 1 ev_comms carrier Make the trades
waitfor ev_pending == 0 30
on 1 waitfor ev_pending == 0 30
waitfor fr_leaving >= 1 10
on 1 expect ev_res_processor_gain == 1

step friendly.riddle
# Spec 1366: the client answers right from its machine. Heat halves in the sim, so both machines see it.
mp heat 40
mp event riddle_hole
waitfor ev_pending == 1 20
mp ev_carrier_goto
wait 2
ev_mark
on 1 ev_mark
on 1 ev_comms carrier A hole.
waitfor ev_pending == 0 30
on 1 waitfor ev_pending == 0 30
expect ev_heat_gain < -15
on 1 expect ev_heat_gain < -15

step friendly.debtor
# Spec 1367, 1368: the client buys a debt from its machine; at its time the debtor pays the client back. Both machines agree.
as 1 contract_money 500000
mp event debtor
waitfor ev_pending == 1 20
mp ev_carrier_goto
wait 2
ev_mark
on 1 ev_mark
on 1 ev_comms carrier Buy their debt
waitfor fr_debtors == 1 30
on 1 waitfor fr_debtors == 1 30
on 1 expect ev_money_gain < 0
mp fr_debtor_end
waitfor fr_debtors == 0 20
on 1 waitfor fr_debtors == 0 30
on 1 expect ev_money_gain == 0
expect ev_money_gain == 0

end
