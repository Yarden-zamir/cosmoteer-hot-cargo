# Co-op factions prices, the blind switch and peace in ally space, host side (spec 644, 650, 655, 1150). Prices are sim state: the client's trades at a fence and at a heat markup
# pass the sim's exact price check (MPMoveResourcesInput) on both machines, with no desync.
mphost {PORT}
mpstart 2
mp ev_set Enabled false
mp godmode on
mp tostation
waitfor mpmark >= 1 180
mp fa_roles - station
mp fa_base
mp mpmark 2
waitfor mpmark >= 3 180
expect fa_rivalsell == 1.5
mp heat 80
mp mpmark 4
waitfor mpmark >= 5 180
expect fa_markup == 1.2
# Spec 1150: the enemy check reads sim state only, so the peace in ally space runs the same on both machines.
mp fa_roles here other
waitfor fa_zone == 1 30
waitfor hunters >= 1 60
mp hunternear 0 600
mp dip_spawn ally 400
mp fa_peace_watch ally
wait 20
expect fa_peace_allyhits == 0
expect fa_peace_hunterhits == 0
expect desyncs == 0
mp mpmark 6
waitfor mpconnected == 0 240
quit
