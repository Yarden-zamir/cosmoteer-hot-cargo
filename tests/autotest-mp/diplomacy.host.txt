# Co-op diplomacy, host side (spec 587): the client hands tech to a hunter and hires a friendly ship through the shared choice
# input. Both machines see the hunters leave, the same vault tech and the hire, with no desync.
mphost {PORT}
mpstart 2
mp godmode on
mp ev_set Enabled false
mp aiset FleeEnabled false
mp aiset CallEnabled false
mp dip_set DemandHailChance 0
mp dip_set MercChance 0
mp dip_set HandOverChanceMin 1
mp dip_set HandOverChanceMax 1
mp dip_set HunterLeaveSeconds 2
mp dip_set HireChanceBase 5
mp mpmark 1
waitfor dip_leaving >= 1 120
expect dip_vaulttech == 6
waitfor hunters == 0 60
waitfor dip_hires >= 1 120
expect dip_friendhired == 1
wait 10
expect desyncs == 0
log host-end
waitfor mpconnected == 0 240
quit
