# Co-op onboarding, host side (spec 721, 733). The co-op page shows, and both machines log the same run log lines.
codex_reset
tutorials on
mphost {PORT}
mpstart {PLAYERS}
mp ev_set Enabled false
wait 8
expect hint_hc_coop == 1
mp heat 30
mp mpmark 1
waitfor mpmark >= 1 60
wait 3
expect runlog_stage == 1
expect runlog_count >= 2
expect desyncs == 0
waitfor mpconnected == 0 240
codex_restore
quit
