# Co-op win, host side: the Buyer appears, the client sells tech to it, and the run is won on both machines.
mphost {PORT}
mpstart {PLAYERS}
mp ev_set Enabled false
mp buyer spawn
waitfor buyers == 1 30
waitfor won == 1 180
expect delivered >= 1
expect score > 0
# Heat stays off after the win.
mp heat -100
wait 10
expect heat == 0
expect desyncs == 0
waitfor mpconnected == 0 240
quit
